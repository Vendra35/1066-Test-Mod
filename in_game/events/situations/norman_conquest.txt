# The opening of the 1066 campaign, fully historical on the AI side: every
# death and every war below happened, on that date. The player is asked, the
# AI is railroaded — alternative options carry `trigger = { is_ai = no }`
# (attested verbatim: events/situations/golden_age_of_piracy.txt:209) and the
# historical path is marked historical_option = yes.
#
# Scheduling comes from the situation's own on_start (the situation owns its
# lifecycle — the on_action route did nothing in game; see KNOWLEDGE.md).
# Offsets are anchored to the observed situation start of 1 October:
# day 0 both wars (hidden retries +1..+13), +3 Stamford Bridge, +13 Hastings (14 Oct), +85 the
# coronation (25 Dec). Event shape cloned from
# events/situations/hundred_years_war.txt.
#
# Trigger law, RESOLVED (2026-07-29, docs/EVENT-SYSTEM.md): the round-2
# "triggers swallow events" hypothesis was FALSE — the triggers were simply
# false. The corrected rule: RAILROAD beats (.1-.90 below) carry NO
# event-level trigger, guards live inside options as if/limit; FLAVOUR
# events (the .100+ DHE band) SHOULD carry triggers, as 95% of vanilla
# DHEs do. The .100+ band uses vanilla's dynamic_historical_event
# self-scheduler (hundred_years_war.txt:452-457's exact shape) — no
# on_start plumbing, fires on its own date window.

# Norway: Hardrada's fleet is already off the English coast.
norman_conquest.11 = {
	type = country_event
	category = situation_event

	title = norman_conquest.11.title
	desc = norman_conquest.11.desc
	outcome = neutral

	image = "gfx/interface/illustrations/situation/_default.dds"

	option = {
		name = norman_conquest.11.a
		historical_option = yes

		# CB first, declaration second: a normal declaration needs the CB in
		# hand (can_declare_war_on inside country_triggers.txt:1198). The
		# old else-branch order cost a month per attempt — measured round 3.
		if = {
			limit = { NOT = { is_at_war_with = c:ENG } }
			leave_all_wars_with = c:ENG
			if = {
				limit = { NOT = { has_casus_belli_on = c:ENG } }
				add_casus_belli = {
					target = c:ENG
					type = casus_belli:cb_norwegian_invasion
				}
			}
			if = {
				limit = {
					can_declare_legal_war_on = { target = c:ENG }
				}
				declare_war_with_cb = { target = c:ENG type = casus_belli:cb_norwegian_invasion }
			}
		}
	}

	option = {
		name = norman_conquest.11.b
		trigger = { is_ai = no }
	}
}

# Norway, +3 days from situation start: Stamford Bridge (historically 25 September).
norman_conquest.12 = {
	type = country_event
	category = situation_event

	title = norman_conquest.12.title
	desc = norman_conquest.12.desc
	outcome = neutral

	image = "gfx/interface/illustrations/situation/_default.dds"

	option = {
		name = norman_conquest.12.a
		historical_option = yes

		# Succession first, so the crown lands on Magnus rather than on a
		# generated heir; then the king falls near York; then the survivors
		# sail home, as they did. The death does NOT depend on the war
		# having formally opened — history is the railroad, the war state
		# is flavour.
		if = {
			limit = {
				character:nor_harald_hardrada ?= { is_alive = yes }
			}
			set_as_designated_heir = { target = character:nor_magnus_ii }
			kill_character = {
				target = character:nor_harald_hardrada
				location = location:york
			}
			leave_all_wars_with = c:ENG
			c:ENG ?= {
				trigger_event_non_silently = { id = norman_conquest.13 }
			}
		}
	}

	option = {
		name = norman_conquest.12.b
		trigger = { is_ai = no }
	}
}

# The first legal day for a declaration is not knowable from script: rounds
# 3 and 4 measured that a day-0 attempt WITH the casus belli in hand still
# fails, and the next monthly retry succeeds. Whether that is a CB
# registration delay or a game-start lock, a short ladder of hidden retries
# converges on the earliest legal day instead of losing a month per try.
# Shape attested verbatim in
# events/ai_area_conqest_events/hidden_events_for_ai_conquest.txt
# (type/outcome/empty_text/hidden, effects in immediate). is_ai-gated: a
# player who chose to hold back stays held back.

norman_conquest.15 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	immediate = {
		if = {
			limit = {
				is_ai = yes
				NOT = { is_at_war_with = c:ENG }
				character:nor_harald_hardrada ?= { is_alive = yes }
			}
			if = {
				limit = { NOT = { has_casus_belli_on = c:ENG } }
				add_casus_belli = {
					target = c:ENG
					type = casus_belli:cb_norwegian_invasion
				}
			}
			if = {
				limit = {
					can_declare_legal_war_on = { target = c:ENG }
				}
				declare_war_with_cb = { target = c:ENG type = casus_belli:cb_norwegian_invasion }
			}
		}
	}
}

# Normandy, +85 days: Westminster Abbey, 25 December. An AI England submits to history;
# a player England is offered the choice in norman_conquest.42 instead —
# a failsafe never takes a player's country.
norman_conquest.40 = {
	type = country_event
	category = situation_event

	title = norman_conquest.40.title
	desc = norman_conquest.40.desc
	outcome = positive

	image = "gfx/interface/illustrations/situation/_default.dds"

	option = {
		name = norman_conquest.40.a
		historical_option = yes

		# Gates live here, not in an event trigger: Harold must be dead and
		# William alive, or nothing happens. The union does not depend on
		# the war having formally opened — see the .12 note.
		if = {
			limit = {
				character:eng_harold_godwinson ?= { is_alive = no }
				character:eng_william_the_conquerer ?= { is_alive = yes }
				c:ENG ?= { is_ai = yes }
			}
			leave_all_wars_with = c:ENG
			force_union = c:ENG
			c:ENG ?= {
				trigger_event_non_silently = { id = norman_conquest.41 }
			}
		}
		else_if = {
			limit = {
				character:eng_harold_godwinson ?= { is_alive = no }
				character:eng_william_the_conquerer ?= { is_alive = yes }
			}
			c:ENG ?= {
				trigger_event_non_silently = { id = norman_conquest.42 }
			}
		}
	}

	option = {
		name = norman_conquest.40.b
		trigger = { is_ai = no }
	}
}
