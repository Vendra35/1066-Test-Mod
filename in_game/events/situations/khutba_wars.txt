######################################
# 10 — the Friday the name changed (MEC, 1071)
######################################

khutba_wars.10 = {
	type = country_event
	category = situation_event

	title = khutba_wars.10.title
	desc = khutba_wars.10.desc
	outcome = neutral

	image = "gfx/interface/illustrations/situation/_default.dds"

	fire_only_once = yes
	dynamic_historical_event = {
		tag = MEC
		from = 1071.4.1
		to = 1071.8.1
		monthly_chance = 100
	}

	trigger = {
		is_subject_of = c:FAT
		country_exists = c:ABS
	}

	option = {
		name = khutba_wars.10.a
		historical_option = yes
		c:FAT ?= { cancel_subject = c:MEC }
		make_subject_of = { target = c:ABS type = subject_type:tributary }
		add_country_modifier = {
			modifier = kw_khutba_for_baghdad
			years = 6
			mode = add
		}
		c:ABS ?= {
			add_country_modifier = {
				modifier = kw_the_caliphs_good_day
				years = 4
				mode = add
			}
			trigger_event_non_silently = { id = khutba_wars.20 days = { 30 60 } }
		}
		c:FAT ?= {
			add_country_modifier = {
				modifier = kw_the_shamed_caliphate
				years = 4
				mode = add
			}
			trigger_event_non_silently = { id = khutba_wars.11 days = { 3 5 } }
		}
		c:YEM ?= {
			trigger_event_non_silently = { id = khutba_wars.30 days = { 20 40 } }
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = khutba_wars.10.b
		add_stability = stability_mild_bonus
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = khutba_wars.10.c
		c:FAT ?= { cancel_subject = c:MEC }
		add_prestige = prestige_mild_bonus
		ai_chance = {
			base = 0
		}
	}
}

	}

		}
