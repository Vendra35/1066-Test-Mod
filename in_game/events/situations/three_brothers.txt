######################################
# 20 — the fall of Galicia: THE DECLARATION-LOCK PROBE (CAS, 1071)
######################################

three_brothers.20 = {
	type = country_event
	category = situation_event

	title = three_brothers.20.title
	desc = three_brothers.20.desc
	outcome = neutral

	image = "gfx/interface/illustrations/situation/_default.dds"

	fire_only_once = yes
	dynamic_historical_event = {
		tag = CAS
		from = 1071.1.1
		to = 1071.7.1
		monthly_chance = 100
	}

	trigger = {
		country_exists = c:GLC
		NOT = { is_at_war_with = c:GLC }
	}

	option = {
		name = three_brothers.20.a
		historical_option = yes
		# THE MEASUREMENT: the CB has been in hand since 1068. Note in
		# game whether the war opens on THIS day or the next monthly
		# tick — that answer is the round-5 lock's scope.
		if = {
			limit = {
				can_declare_legal_war_on = { target = c:GLC }
			}
			declare_war_with_cb = { target = c:GLC type = casus_belli:cb_claim_throne }
		}
		ai_chance = {
			base = 100
		}
	}

	option = {
		name = three_brothers.20.b
		set_variable = {
			name = tb_galicia_spared
			years = 10
		}
		add_stability = stability_mild_bonus
		ai_chance = {
			base = 0
		}
	}
}
